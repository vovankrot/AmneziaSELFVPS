cmake_minimum_required(VERSION 3.25)
project(CdnWfpControl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
set(CMAKE_AUTOMOC ON)
add_executable(cdn-wfp-regressions tests.cpp)
target_include_directories(cdn-wfp-regressions PRIVATE ../../client)
target_link_libraries(cdn-wfp-regressions PRIVATE Qt6::Core Qt6::Network Qt6::Test fwpuclnt iphlpapi ws2_32)
add_executable(cdn-wfp-control control.cpp)
target_include_directories(cdn-wfp-control PRIVATE ../../client)
target_link_libraries(cdn-wfp-control PRIVATE Qt6::Core Qt6::Network fwpuclnt iphlpapi ws2_32)
enable_testing()
add_test(NAME cdn-wfp-scoped-transactions COMMAND cdn-wfp-regressions)
