cmake_minimum_required(VERSION 3.25)
project(ProcessPolicyRegressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
foreach(INPUT ../../client/utilities.cpp ../../ipc/ipcserverprocess.cpp)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${INPUT})
endforeach()
file(READ ../../client/utilities.cpp SOURCE)
string(FIND "${SOURCE}" "bool Utils::killProcessByName(" START)
string(FIND "${SOURCE}" "QString Utils::openVpnExecPath()" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} METHOD)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-kill-process.inc "${METHOD}")
file(READ ../../ipc/ipcserverprocess.cpp SOURCE)
string(FIND "${SOURCE}" "void IpcServerProcess::start()" START)
string(FIND "${SOURCE}" "void IpcServerProcess::terminate()" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} METHOD)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-start-process.inc "${METHOD}")
string(FIND "${SOURCE}" "bool IpcServerProcess::waitForFinished()" START)
string(SUBSTRING "${SOURCE}" ${START} -1 TAIL)
string(FIND "${TAIL}" "\n#endif" LENGTH)
string(SUBSTRING "${TAIL}" 0 ${LENGTH} METHOD)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-wait-process.inc "${METHOD}")
add_executable(process-policy-regressions main.cpp)
target_include_directories(process-policy-regressions PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ../..)
target_link_libraries(process-policy-regressions PRIVATE Qt6::Core Qt6::Test)
add_executable(selfvps-process-policy-fixture fixture.cpp)
target_link_libraries(selfvps-process-policy-fixture PRIVATE Qt6::Core)
add_dependencies(process-policy-regressions selfvps-process-policy-fixture)
enable_testing()
add_test(NAME process-path-isolation COMMAND process-policy-regressions)
