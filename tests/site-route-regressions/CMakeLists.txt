cmake_minimum_required(VERSION 3.25)
project(SiteRouteRegressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network RemoteObjects Test)
set(CONNECTION ../../client/vpnconnection.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CONNECTION})
file(READ ${CONNECTION} SOURCE)
string(FIND "${SOURCE}" "void VpnConnection::addSitesRoutes(" START)
string(FIND "${SOURCE}" "QSharedPointer<VpnProtocol> VpnConnection::vpnProtocol()" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} METHOD)
# Keep the production method, substitute only asynchronous DNS completion.
string(REPLACE "QHostInfo::lookupHost" "TestDns::lookupHost" METHOD "${METHOD}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-site-routes.inc "${METHOD}")
string(FIND "${SOURCE}" "void VpnConnection::shutdown()" CLEAN_START)
string(FIND "${SOURCE}" "void VpnConnection::setConnectionState(" CLEAN_END)
math(EXPR CLEAN_LENGTH "${CLEAN_END} - ${CLEAN_START}")
string(SUBSTRING "${SOURCE}" ${CLEAN_START} ${CLEAN_LENGTH} CLEAN_METHODS)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-session-cleanup.inc "${CLEAN_METHODS}")
string(FIND "${SOURCE}" "void VpnConnection::createProtocolConnections()" PROTOCOL_START)
string(SUBSTRING "${SOURCE}" ${PROTOCOL_START} -1 PROTOCOL_TAIL)
string(FIND "${PROTOCOL_TAIL}" "    connect(m_vpnProtocol.data(), SIGNAL(bytesChanged" PROTOCOL_END)
string(SUBSTRING "${PROTOCOL_TAIL}" 0 ${PROTOCOL_END} PROTOCOL_METHOD)
string(REPLACE "void VpnConnection::createProtocolConnections()" "void VpnConnection::connectProtocolCleanupSignals()" PROTOCOL_METHOD "${PROTOCOL_METHOD}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-protocol-cleanup-signals.inc "${PROTOCOL_METHOD}\n}\n")
string(FIND "${SOURCE}" "void VpnConnection::createProtocolConnections()" START)
string(SUBSTRING "${SOURCE}" ${START} -1 TAIL)
string(FIND "${TAIL}" "#ifdef AMNEZIA_DESKTOP" START)
string(FIND "${TAIL}" "#endif" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${TAIL}" ${START} ${LENGTH} CONNECTIONS)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-service-signals.inc
    "void VpnConnection::connectServiceSignals() {\n${CONNECTIONS}\n#endif\n}\n")
string(FIND "${SOURCE}" "void VpnConnection::onKillSwitchModeChanged(" KS_START)
string(FIND "${SOURCE}" "void VpnConnection::onConnectionStateChanged(" KS_END)
math(EXPR KS_LENGTH "${KS_END} - ${KS_START}")
string(SUBSTRING "${SOURCE}" ${KS_START} ${KS_LENGTH} KS_METHOD)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-strict-ks.inc "${KS_METHOD}")
add_executable(site-route-regressions main.cpp)
qt_add_repc_sources(site-route-regressions service.rep)
qt_add_repc_replicas(site-route-regressions service.rep)
target_compile_definitions(site-route-regressions PRIVATE AMNEZIA_DESKTOP)
target_include_directories(site-route-regressions PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ../..)
target_link_libraries(site-route-regressions PRIVATE Qt6::Core Qt6::Network Qt6::RemoteObjects Qt6::Test)
enable_testing()
add_test(NAME site-route-session-lifetime COMMAND site-route-regressions)
string(FIND "${SOURCE}" "void VpnConnection::appendXrayRoutingConfig()" DNS_START)
string(FIND "${SOURCE}" "#ifdef Q_OS_ANDROID\nvoid VpnConnection::restoreConnection()" DNS_END)
math(EXPR DNS_LENGTH "${DNS_END} - ${DNS_START}")
string(SUBSTRING "${SOURCE}" ${DNS_START} ${DNS_LENGTH} DNS_METHOD)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-xray-dns.inc "${DNS_METHOD}")
add_executable(xray-dns-regressions xray-dns.cpp)
target_include_directories(xray-dns-regressions PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(xray-dns-regressions PRIVATE Qt6::Core Qt6::Network Qt6::Test)
add_test(NAME xray-dns-rule-priority COMMAND xray-dns-regressions)

string(FIND "${SOURCE}" "void VpnConnection::startConfiguredProtocol()" PREPARE_END)
math(EXPR PREPARE_LENGTH "${PREPARE_END} - ${PREPARE_START}")
string(SUBSTRING "${SOURCE}" ${PREPARE_START} ${PREPARE_LENGTH} PREPARE_METHODS)
string(REPLACE "VpnConnection::" "PreparedConnection::" PREPARE_METHODS "${PREPARE_METHODS}")
