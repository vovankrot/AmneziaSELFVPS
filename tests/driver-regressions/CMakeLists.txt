# Execute production driver shutdown and native route methods without OS writes.
function(extract_method SOURCE_FILE MARKER END_MARKER OUTPUT)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SOURCE_FILE})
    file(READ ${SOURCE_FILE} BODY)
    string(FIND "${BODY}" "${MARKER}" START)
    string(SUBSTRING "${BODY}" ${START} -1 TAIL)
    string(FIND "${TAIL}" "${END_MARKER}" END)
    string(SUBSTRING "${TAIL}" 0 ${END} METHOD)
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT} "${METHOD}")
endfunction()

# Resolve Qt from the same installation used for linking, independently of PATH.
# Other installed tools can contribute an incompatible Qt6Core.dll to PATH.
get_property(REGRESSION_TARGETS DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(REGRESSION_TARGET IN LISTS REGRESSION_TARGETS)
    add_custom_command(TARGET ${REGRESSION_TARGET} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:Qt6::Core> $<TARGET_FILE:Qt6::Network>
            $<TARGET_FILE_DIR:${REGRESSION_TARGET}>)
endforeach()
