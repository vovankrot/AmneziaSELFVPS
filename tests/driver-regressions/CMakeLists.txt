cmake_minimum_required(VERSION 3.25)
project(DriverRegressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
# Exercise the production helper, substituting only the OS call and log sink.
file(READ ../../client/platforms/windows/daemon/windowssplittunnel.cpp SOURCE)
string(FIND "${SOURCE}" "constexpr int kIoctlTimeoutMs" START)
string(FIND "${SOURCE}" "// App paths reach us" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} HELPER)
string(REPLACE "kIoctlTimeoutMs = 1500" "kIoctlTimeoutMs = 40" HELPER "${HELPER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production-helper.h" "${HELPER}")
add_executable(driver-regressions main.cpp)
target_include_directories(driver-regressions PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../client/core")
enable_testing()
add_test(NAME driver-failure-lifetime COMMAND driver-regressions)

add_executable(socket-binding-regressions socket-binding.cpp)
target_include_directories(socket-binding-regressions PRIVATE ../../service/server)
target_link_libraries(socket-binding-regressions PRIVATE ws2_32)
add_test(NAME socket-interface-family COMMAND socket-binding-regressions)
