# Execute the model's production list/selection methods against isolated storage.
function(extract_server_method SOURCE_FILE START END OUTPUT)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SOURCE_FILE})
  file(READ ${SOURCE_FILE} SOURCE)
  string(FIND "${SOURCE}" "${START}" FIRST)
  string(FIND "${SOURCE}" "${END}" LAST)
  if(FIRST LESS 0 OR LAST LESS FIRST)
    message(FATAL_ERROR "Server model method boundary not found: ${START}")
  endif()
  math(EXPR LENGTH "${LAST} - ${FIRST}")
  string(SUBSTRING "${SOURCE}" ${FIRST} ${LENGTH} METHOD)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT} "${METHOD}")
endfunction()
