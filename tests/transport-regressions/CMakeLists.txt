cmake_minimum_required(VERSION 3.25)
project(TransportRegressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Core5Compat Network Concurrent RemoteObjects Test)
add_executable(transport-regressions main.cpp)
qt_add_repc_replicas(transport-regressions ipc_process_interface.rep)
qt_add_repc_sources(transport-regressions ipc_process_interface.rep)
qt_add_repc_replicas(transport-regressions ipc_interface.rep)
qt_add_repc_sources(transport-regressions ipc_interface.rep)
target_sources(transport-regressions PRIVATE ../../client/core/networkUtilities.cpp ../../client/core/networkUtilities.h)
target_include_directories(transport-regressions PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ../.. ../../client)
target_include_directories(transport-regressions PRIVATE ../../client/3rd-prebuilt/3rd-prebuilt/openssl/windows/include)
target_link_libraries(transport-regressions PRIVATE Qt6::Core Qt6::Core5Compat Qt6::Network Qt6::Concurrent Qt6::RemoteObjects Qt6::Test
    ${CMAKE_CURRENT_SOURCE_DIR}/../../client/3rd-prebuilt/3rd-prebuilt/openssl/windows/win64/libcrypto.lib iphlpapi ws2_32 crypt32)
enable_testing()
add_test(NAME transport-regressions COMMAND transport-regressions)
add_executable(route-probe route-probe.cpp ../../client/core/networkUtilities.cpp ../../client/core/networkUtilities.h)
target_include_directories(route-probe PRIVATE ../.. ../../client)
target_link_libraries(route-probe PRIVATE Qt6::Core Qt6::Core5Compat Qt6::Network iphlpapi ws2_32)
