cmake_minimum_required(VERSION 3.25)
project(update-regressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Test)
configure_file(../../version.h.in version.h)
add_executable(update-regressions main.cpp ../../client/core/appUpdater.cpp ../../client/core/appUpdater.h)
target_compile_definitions(update-regressions PRIVATE APP_VERSION="5.0.0.8")
target_include_directories(update-regressions PRIVATE ../.. ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(update-regressions PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
if(WIN32)
  target_link_libraries(update-regressions PRIVATE shell32)
endif()
