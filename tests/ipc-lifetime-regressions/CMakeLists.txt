cmake_minimum_required(VERSION 3.25)
project(IpcLifetimeRegressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network RemoteObjects Test)
set(SERVER ../../ipc/ipcserver.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SERVER})
file(READ ${SERVER} SOURCE)
string(FIND "${SOURCE}" "int IpcServer::createPrivilegedProcess()" START)
string(FIND "${SOURCE}" "int IpcServer::routeAddList" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} METHODS)
string(REPLACE "singleShot(30000" "singleShot(40" METHODS "${METHODS}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-process-lifetime.inc "${METHODS}")
add_executable(ipc-lifetime-regressions main.cpp)
qt_add_repc_replicas(ipc-lifetime-regressions process.rep)
qt_add_repc_sources(ipc-lifetime-regressions process.rep)
target_include_directories(ipc-lifetime-regressions PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(ipc-lifetime-regressions PRIVATE Qt6::Core Qt6::Network Qt6::RemoteObjects Qt6::Test)
add_custom_command(TARGET ipc-lifetime-regressions POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:Qt6::Core> $<TARGET_FILE:Qt6::Network> $<TARGET_FILE:Qt6::RemoteObjects> $<TARGET_FILE:Qt6::Test>
        $<TARGET_FILE_DIR:ipc-lifetime-regressions>)
enable_testing()
add_test(NAME ipc-process-lifetime COMMAND ipc-lifetime-regressions)
