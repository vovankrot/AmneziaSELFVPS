cmake_minimum_required(VERSION 3.25)
project(ClientRefreshRegressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Concurrent Test)
# Exercise the exact production orchestration with controlled SSH/model doubles.
set(CONTROLLER ${CMAKE_CURRENT_SOURCE_DIR}/../../client/ui/controllers/exportController.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CONTROLLER})
file(READ ${CONTROLLER} SOURCE)
string(FIND "${SOURCE}" "void ExportController::updateClientManagementModel(" START)
string(FIND "${SOURCE}" "void ExportController::revokeConfig(" END)
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} FUNCTION)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/production-refresh.inc "${FUNCTION}")
add_executable(client-refresh-regressions main.cpp)
target_include_directories(client-refresh-regressions PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(client-refresh-regressions PRIVATE Qt6::Core Qt6::Concurrent Qt6::Test)
enable_testing()
add_test(NAME client-refresh-regressions COMMAND client-refresh-regressions)
